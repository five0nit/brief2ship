# Brief2Ship v0.3.1 Install & Trust Implementation Plan

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Make the existing Brief2Ship standard accurate, natively installable, reproducibly checked, and backed by one real worked example.

**Architecture:** Keep Markdown and `skills/brief2ship/SKILL.md` as the product. Replace the maintainer-specific copy script with Hermes’ supported URL installer, add lightweight cross-file validation and CI, and promote the existing Brief Checker dogfood as a bounded example rather than expanding the core workflow.

**Release:** `0.3.1 — Install & Trust`

import re
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]


class ReleaseContractTests(unittest.TestCase):
    def test_version_surfaces_match(self):
        version = (ROOT / "VERSION").read_text(encoding="utf-8").strip()
        self.assertRegex(version, r"^\d+\.\d+\.\d+$")

        skill = (ROOT / "skills/brief2ship/SKILL.md").read_text(encoding="utf-8")
        skill_version = re.search(r"^version:\s*(\S+)\s*$", skill, re.MULTILINE)
        self.assertIsNotNone(skill_version)
        self.assertEqual(version, skill_version.group(1))

        changelog = (ROOT / "CHANGELOG.md").read_text(encoding="utf-8")
        self.assertIn(f"## {version} -", changelog)

    def test_skill_frontmatter_has_required_fields(self):
        skill = (ROOT / "skills/brief2ship/SKILL.md").read_text(encoding="utf-8")
        self.assertTrue(skill.startswith("---\n"))
        frontmatter = skill.split("---", 2)[1]
        for key in ("name:", "description:", "version:", "license:", "platforms:"):
            self.assertIn(key, frontmatter)

    def test_license_names_holder(self):
        license_text = (ROOT / "LICENSE").read_text(encoding="utf-8")
        self.assertIn("Michael Costea", license_text)

    def test_public_files_have_no_maintainer_paths(self):
        paths = [
            ROOT / "README.md",
            ROOT / "CONTRIBUTING.md",
            ROOT / "CHANGELOG.md",
            ROOT / "docs/how-it-works.md",
            ROOT / "docs/lanes.md",
            ROOT / "docs/report-document-lane.md",
            ROOT / "docs/examples.md",
            ROOT / "skills/brief2ship/SKILL.md",
            *sorted((ROOT / "templates").glob("*.md")),
            *sorted((ROOT / "scripts").glob("*")),
        ]
        for path in paths:
            if not path.is_file():
                continue
            text = path.read_text(encoding="utf-8")
            self.assertNotIn("<maintainer-home>", text, path)
            self.assertNotIn("profiles/<maintainer-profile>", text, path)

    def test_unsupported_alias_claim_is_absent(self):
        readme = (ROOT / "README.md").read_text(encoding="utf-8")
        self.assertNotIn("Backward-compatible skill names", readme)
        self.assertNotIn("Hermes/OpenClaw skill package", readme)

    def test_report_lane_is_discoverable(self):
        skill = (ROOT / "skills/brief2ship/SKILL.md").read_text(encoding="utf-8")
        frontmatter = skill.split("---", 2)[1].lower()
        trigger = "\n".join(skill.splitlines()[12:20]).lower()
        self.assertIn("report", frontmatter)
        self.assertIn("report", trigger)

    def test_four_lane_contract_matches_skill(self):
        skill = (ROOT / "skills/brief2ship/SKILL.md").read_text(encoding="utf-8")
        lanes = (ROOT / "docs/lanes.md").read_text(encoding="utf-8")
        for name in ("App", "Dashboard / Internal tool", "Landing page", "Report / Document"):
            self.assertIn(name, skill)
            self.assertIn(name, lanes)

    def test_readme_relative_links_exist(self):
        readme = (ROOT / "README.md").read_text(encoding="utf-8")
        for raw in re.findall(r"\[[^]]+\]\(([^)]+)\)", readme):
            if "://" in raw or raw.startswith("#"):
                continue
            target = raw.split("#", 1)[0]
            self.assertTrue((ROOT / target).exists(), target)

    def test_public_install_uses_native_hermes_cli(self):
        readme = (ROOT / "README.md").read_text(encoding="utf-8")
        self.assertIn("hermes skills install", readme)
        self.assertIn("hermes --profile <name> skills install", readme)
        self.assertIn("/v0.3.1/skills/brief2ship/SKILL.md", readme)
        self.assertFalse((ROOT / "scripts/install-skill.sh").exists())

**Step 1: Implement the validator**

Create `scripts/validate-release.py`:

```python
#!/usr/bin/env python3
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
COMMANDS = [
    [sys.executable, str(ROOT / "scripts/validate-docs.py")],
    [
        sys.executable,
        "-m",
        "unittest",
        "discover",
        "-s",
        str(ROOT / "tests"),
        "-p",
        "test_*.py",
        "-v",
    ],
]


def main() -> int:
    for command in COMMANDS:
        result = subprocess.run(command, cwd=ROOT, check=False)
        if result.returncode:
            return result.returncode
    print("Brief2Ship release validation passed")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
